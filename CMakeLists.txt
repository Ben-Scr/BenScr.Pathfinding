cmake_minimum_required(VERSION 3.28)
project(BenScrPathfinding VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
endif()

option(BENSCR_PATHFINDING_BUILD_TESTS "Build contract tests" ${PROJECT_IS_TOP_LEVEL})
option(BENSCR_PATHFINDING_BUILD_EXAMPLES "Build example" ${PROJECT_IS_TOP_LEVEL})
option(BENSCR_PATHFINDING_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(benscr_pathfinding STATIC src/grid.cpp)
add_library(BenScr::Pathfinding ALIAS benscr_pathfinding)
set_target_properties(benscr_pathfinding PROPERTIES EXPORT_NAME Pathfinding CXX_EXTENSIONS OFF)
target_compile_features(benscr_pathfinding PUBLIC cxx_std_20)
target_include_directories(benscr_pathfinding PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)

function(benscr_pathfinding_checks target)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(BENSCR_PATHFINDING_SANITIZERS)
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        else()
            message(FATAL_ERROR "Sanitizers require GCC or Clang on a supported POSIX host.")
        endif()
    endif()
endfunction()
benscr_pathfinding_checks(benscr_pathfinding)

if(BUILD_TESTING AND BENSCR_PATHFINDING_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(pathfinding_contract_tests tests/pathfinding_tests.cpp)
    target_link_libraries(pathfinding_contract_tests PRIVATE BenScr::Pathfinding Threads::Threads)
    benscr_pathfinding_checks(pathfinding_contract_tests)
    add_test(NAME pathfinding_contract COMMAND pathfinding_contract_tests)
    add_executable(pathfinding_header_test tests/header.cpp)
    target_link_libraries(pathfinding_header_test PRIVATE BenScr::Pathfinding)
    benscr_pathfinding_checks(pathfinding_header_test)
    add_test(NAME standalone_header COMMAND pathfinding_header_test)
endif()
if(BENSCR_PATHFINDING_BUILD_EXAMPLES)
    add_executable(pathfinding_example examples/pathfinding.cpp)
    target_link_libraries(pathfinding_example PRIVATE BenScr::Pathfinding)
    benscr_pathfinding_checks(pathfinding_example)
endif()

install(TARGETS benscr_pathfinding EXPORT BenScrPathfindingTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE.txt README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/BenScrPathfinding)
configure_package_config_file(cmake/BenScrPathfindingConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/BenScrPathfindingConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrPathfinding)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/BenScrPathfindingConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BenScrPathfindingConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/BenScrPathfindingConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrPathfinding)
install(EXPORT BenScrPathfindingTargets NAMESPACE BenScr:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrPathfinding)
